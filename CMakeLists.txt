cmake_minimum_required(VERSION 3.16)
project(tsp_simulated_annealing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(tsp_sa src/main.cpp)
if(WIN32)
  target_link_libraries(tsp_sa PRIVATE gdi32 user32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_cpp tests/test_cpp.cpp)
  add_test(NAME solver COMMAND test_cpp)
  set_tests_properties(solver PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
